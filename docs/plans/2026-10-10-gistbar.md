# Plan: gistbar

**Spec:** `docs/specs/gistbar/SPEC.md` · **Depth:** Deep · **Tasks:** 16 · **Date:** 2026-10-10

Noctalia Shell v4 plugin for the user's GitHub gists through the `gh` CLI. Same
architecture and look as Daybar (`~/dev/projects/noctalia-daybar`): logic in
`utils/**/*.js` QML JavaScript libraries (`.pragma library`, `.import`, top-level
declarations at column 0, no ES `import`/`export`), tested with vitest through the
`qml-library-to-esm` transform; thin QML; Daybar's `components/ui/` kit copied.

**Global rules for every task**

- TDD for `utils/`: test first, see it fail, implement. `npm test -- tests/<f>.test.js`,
  then `npm test` (read `Tests  N passed`).
- Every external command is an argv array. Gist content, JSON bodies and clipboard
  writes go through `Process.write()` + `stdinEnabled = false` (EOF), never argv.
- Never write `/home/<user>` paths in files (fixtures use `/data/u`); never log gist
  contents.
- English code/comments; UI strings via `pluginApi?.tr()` in both i18n files.
- No git commands inside tasks (the orchestrator commits).

**Implement** `utils/gists.js`
```js
.pragma library

// gh api JSON -> normalized gists. List items have no content; details do.

function filesOf(files, withContent) {
  return Object.keys(files || {}).map((k) => {
    const f = files[k] || {};
    const base = { name: String(f.filename || k), language: String(f.language || "") };
    if (withContent) return Object.assign(base, { content: String(f.content || ""), truncated: !!f.truncated });
    return Object.assign(base, { size: Number(f.size) || 0 });
  });
}

function normalize(g) {
  return {
    id: String(g.id),
    url: String(g.html_url || ""),
    description: String(g.description || ""),
    public: !!g.public,
    updatedAt: String(g.updated_at || ""),
    files: filesOf(g.files, false)
  };
}

function byUpdatedDesc(a, b) {
  return a.updatedAt < b.updatedAt ? 1 : a.updatedAt > b.updatedAt ? -1 : 0;
}

// `gh api gists --paginate --slurp` -> [[page], [page]]; a plain page also works.
function parseList(text) {
  let o;
  try {
    o = JSON.parse(String(text || ""));
  } catch (e) {
    return { ok: false, error: "parse" };
  }
  if (!Array.isArray(o)) return { ok: false, error: "parse" };
  const items = o.length > 0 && Array.isArray(o[0]) ? [].concat.apply([], o) : o;
  const gists = items.filter((g) => g && g.id !== undefined && g.id !== null).map(normalize);
  gists.sort(byUpdatedDesc);
  return { ok: true, gists: gists };
}

function parseDetail(text) {
  let g;
  try {
    g = JSON.parse(String(text || ""));
  } catch (e) {
    return { ok: false, error: "parse" };
  }
  if (!g || !g.id) return { ok: false, error: "parse" };
  return { ok: true, gist: normalize(g), entry: { id: String(g.id), updatedAt: String(g.updated_at || ""), files: filesOf(g.files, true) } };
}

function title(g) {
  return g.description || (g.files[0] ? g.files[0].name : g.id);
}

function prepend(list, gist) {
  return [gist].concat(list.filter((g) => g.id !== gist.id));
}

function remove(list, id) {
  return list.filter((g) => g.id !== id);
}
```

**Implement** `utils/search.js`
```js
.pragma library
.import "text.js" as Text

// Rank: 0 = description or file name, 1 = language, 2 = content only.
function rank(g, entry, query) {
  const names = [g.description].concat(g.files.map((f) => f.name));
  if (Text.matches(names, query)) return 0;
  const langs = names.concat(g.files.map((f) => f.language));
  if (Text.matches(langs, query)) return 1;
  const body = entry ? entry.files.map((f) => f.content) : [];
  return Text.matches(langs.concat(body), query) ? 2 : -1;
}

function visible(g, filter) {
  return filter === "public" ? g.public : filter === "secret" ? !g.public : true;
}

// gists are already sorted by updatedAt desc; the sort below is stable.
function search(gists, contents, query, filter) {
  const pool = gists.filter((g) => visible(g, filter));
  if (String(query || "").trim() === "") return pool;
  return pool
    .map((g, i) => ({ g: g, i: i, r: rank(g, (contents || {})[g.id], query) }))
    .filter((x) => x.r >= 0)
    .sort((a, b) => a.r - b.r || a.i - b.i)
    .map((x) => x.g);
}

function counts(gists) {
  const pub = gists.filter((g) => g.public).length;
  return { all: gists.length, public: pub, secret: gists.length - pub };
}
```

```js
.pragma library

var SHEBANG = [[/\b(bash|sh|zsh)\b/, "sh"], [/\bpython/, "py"], [/\bnode\b/, "js"], [/\bruby\b/, "rb"]];

function suggestName(text) {
  const s = String(text || "").trim();
  if (s.indexOf("#!") === 0) {
    const first = s.split("\n")[0];
    for (const [re, ext] of SHEBANG) if (re.test(first)) return "script." + ext;
    return "script.sh";
  }
  if (s.charAt(0) === "{" || s.charAt(0) === "[") {
    try {
      JSON.parse(s);
      return "data.json";
    } catch (e) {}
  }
  if (/^(#{1,6}\s|[-*]\s)/.test(s)) return "notes.md";
  if (/^<[a-zA-Z!]/.test(s)) return "snippet.html";
  return "snippet.txt";
}
```

```js
.pragma library

// argv builders for the gh CLI. Payloads (content, JSON) always go on stdin.

var ID = /^[0-9a-zA-Z]+$/;

function checkId(id) {
  if (!ID.test(String(id))) throw new Error("invalid gist id");
  return String(id);
}

function listArgs() {
  return ["gh", "api", "gists?per_page=100", "--paginate", "--slurp"];
}

function detailArgs(id) {
  return ["gh", "api", "gists/" + checkId(id)];
}

function deleteArgs(id) {
  return ["gh", "gist", "delete", checkId(id), "--yes"];
}

function createArgs(filename, description, isPublic) {
  const args = ["gh", "gist", "create", "--filename", String(filename), "--desc", String(description || "")];
  if (isPublic) args.push("--public");
  args.push("-");
  return args;
}

function patchArgs(id) {
  return ["gh", "api", "--method", "PATCH", "gists/" + checkId(id), "--input", "-"];
}

function patchBody(description, filename, content) {
  const files = {};
  files[filename] = { content: String(content) };
  return JSON.stringify({ description: String(description || ""), files: files });
}

// `gh gist create` prints the gist URL on stdout.
function createdId(stdout) {
  const m = /gist\.github\.com\/[^/\s]+\/([0-9a-zA-Z]+)/.exec(String(stdout || ""));
  return m ? m[1] : "";
}

// exit code + stderr -> "" | "gh-login" | "gh-missing" | "network" | "failed"
function classify(code, stderr) {
  if (code === 0) return "";
  const e = String(stderr || "");
  if (code === 127) return "gh-missing";
  if (code === 4 || /gh auth login|HTTP 401|Bad credentials/i.test(e)) return "gh-login";
  if (/error connecting|timeout|could not resolve|network/i.test(e)) return "network";
  return "failed";
}
```

**Implement** `utils/cachesync.js`
```js
.pragma library

// cached: { id: updatedAt } read from the cache directory.
function plan(list, cached) {
  const have = cached || {};
  const listed = {};
  const fetch = [];
  for (const g of list) {
    listed[g.id] = true;
    if (have[g.id] !== g.updatedAt) fetch.push(g.id);
  }
  const drop = Object.keys(have).filter((id) => !listed[id]);
  return { fetch: fetch, drop: drop };
}

function createQueue() {
  return { pending: [], running: [] };
}

function enqueue(q, ids) {
  const pending = q.pending.slice();
  for (const id of ids) if (pending.indexOf(id) < 0 && q.running.indexOf(id) < 0) pending.push(id);
  return { pending: pending, running: q.running.slice() };
}

function nextJobs(q, limit) {
  const free = Math.max(0, limit - q.running.length);
  const start = q.pending.slice(0, free);
  return { start: start, queue: { pending: q.pending.slice(free), running: q.running.concat(start) } };
}

function finish(q, id) {
  return { pending: q.pending.slice(), running: q.running.filter((x) => x !== id) };
}
```

```js
function relative(iso, nowMs) {
```
